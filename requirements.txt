# 2D physics engine for the physics/ climber simulation. Pure-Python wrapper
# around Chipmunk2D. Chosen over pybox2d because pymunk's wheels are
# reliably built, the API is cleaner, and it integrates well with
# Gymnasium-based RL environments.
pymunk==7.0.1
# Gymnasium is the maintained successor to OpenAI Gym. The 2D and 3D
# RL environments follow the standard reset()/step() API.
gymnasium==1.0.0
# 3D physics for the sim3d/ climber simulation. MuJoCo is
# Apache-2 since 2021, has fast contact solving, native Python
# bindings, an interactive viewer, and an MJX/JAX backend that the
# RL stack can scale onto a GPU later. The 3.x line ships
# manylinux wheels so no native build is needed inside Docker.
mujoco==3.2.6

# PPO training/replay support for rl/ and sim3d/. This pulls PyTorch as
# a transitive dependency; keep it pinned so saved-policy compatibility
# and Docker rebuilds are reproducible.

# TensorBoard event viewer used by sim3d.train when logging PPO runs.
